 RFM BASED CUSTOMER SEGMENTATION ANALYSIS
   SQL QUERIES USED


 1. CREATE DATABASE :
CREATE DATABASE customer_analysis;
USE customer_analysis;


 2. TOP CUSTOMERS BY REVENUE:
SELECT `Customer ID`,
SUM(Quantity * Price) AS TotalSpent
FROM sales
GROUP BY `Customer ID`
ORDER BY TotalSpent DESC
LIMIT 10;


3. REVENUE BY COUNTRY:
SELECT  Country,
SUM(Quantity * Price) AS Revenue
FROM sales
GROUP BY Country
ORDER BY Revenue DESC;

 4. MONTHLY REVENUE TREND:
SELECT  MONTH(InvoiceDate) AS MonthNo,
SUM(Quantity * Price) AS Revenue
FROM sales
GROUP BY MonthNo
ORDER BY MonthNo;

 5. TOP PRODUCTS BY REVENUE:
SELECT  Description,
SUM(Quantity * Price) AS ProductRevenue
FROM sales
GROUP BY Description
ORDER BY ProductRevenue DESC
LIMIT 10;

 6. TOTAL ORDERS BY COUNTRY:
SELECT Country,
COUNT(DISTINCT Invoice) AS TotalOrders
FROM sales
GROUP BY Country
ORDER BY TotalOrders DESC;

 7. AVERAGE ORDER VALUE:
SELECT AVG(Quantity * Price) AS AvgOrderValue
FROM sales;

 8. MONTHLY CUSTOMER COUNT:
SELECT MONTH(InvoiceDate) AS MonthNo,
 COUNT(DISTINCT `Customer ID`) AS TotalCustomers
FROM sales
GROUP BY MonthNo
ORDER BY MonthNo;

 9. TOTAL REVENUE:
SELECT SUM(Quantity * Price) AS TotalRevenue
FROM sales;

 10. TOTAL CUSTOMERS:
SELECT  COUNT(DISTINCT `Customer ID`) AS TotalCustomers
FROM sales;

 11. TOTAL ORDERS:
SELECT COUNT(DISTINCT Invoice) AS TotalOrders
FROM sales;

 12. REPEAT CUSTOMERS:
SELECT  COUNT(*) AS RepeatCustomers
FROM
(
    SELECT `Customer ID`
    FROM sales
    GROUP BY `Customer ID`
    HAVING COUNT(DISTINCT Invoice) > 1
) AS repeat_data;



 13. CUSTOMER PURCHASE FREQUENCY
SELECT `Customer ID`,
 COUNT(DISTINCT Invoice) AS Frequency
FROM sales
GROUP BY `Customer ID`
ORDER BY Frequency DESC;

 14. CUSTOMER MONETARY VALUE
SELECT `Customer ID`,
SUM(Quantity * Price) AS Monetary
FROM sales
GROUP BY `Customer ID`
ORDER BY Monetary DESC;

 15. RECENCY ANALYSIS
SELECT `Customer ID`,
    DATEDIFF(
        MAX(InvoiceDate),
        MIN(InvoiceDate)
    ) AS Recency
FROM sales
GROUP BY `Customer ID`
ORDER BY Recency DESC;

 16. RFM TABLE
SELECT `Customer ID`,
DATEDIFF(
        (SELECT MAX(InvoiceDate) FROM sales),
        MAX(InvoiceDate)
    ) AS Recency,

    COUNT(DISTINCT Invoice) AS Frequency,

    SUM(Quantity * Price) AS Monetary

FROM sales

GROUP BY `Customer ID`;


 17. CUSTOMER SEGMENTATION

SELECT `Customer ID`,
    SUM(Quantity * Price) AS Monetary,
    COUNT(DISTINCT Invoice) AS Frequency,
    CASE
        WHEN SUM(Quantity * Price) > 5000
             AND COUNT(DISTINCT Invoice) > 10
        THEN 'Champions'

        WHEN SUM(Quantity * Price) > 3000
             AND COUNT(DISTINCT Invoice) > 5
        THEN 'Loyal Customers'

        WHEN SUM(Quantity * Price) > 1500
        THEN 'Potential Loyalists'

        WHEN COUNT(DISTINCT Invoice) <= 3
        THEN 'At Risk'

        ELSE 'Lost Customers'

    END AS CustomerSegment

FROM sales

GROUP BY `Customer ID`;

18. RETENTION RATE
SELECT
(
    COUNT(DISTINCT CASE
        WHEN InvoiceCount > 1
        THEN `Customer ID`
    END)
    /
    COUNT(DISTINCT `Customer ID`)
) * 100 AS RetentionRate

FROM
(
    SELECT
        `Customer ID`,
        COUNT(DISTINCT Invoice) AS InvoiceCount
    FROM sales
    GROUP BY `Customer ID`
) AS retention_table;


 19. CHURN RISK CUSTOMERS

SELECT
    `Customer ID`,
    COUNT(DISTINCT Invoice) AS Orders
FROM sales
GROUP BY `Customer ID`
HAVING Orders <= 2;


 20. CUSTOMER LIFETIME VALUE (CLV)
SELECT
    `Customer ID`,
    SUM(Quantity * Price) AS CLV
FROM sales
GROUP BY `Customer ID`
ORDER BY CLV DESC;
